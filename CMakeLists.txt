cmake_minimum_required(VERSION 3.24)
project(MicrotonalPitchMapPro VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/bigobj /utf-8)
endif()
include(FetchContent)
set(JUCE_DIR "" CACHE PATH "Existing JUCE 8.0.12 checkout")
set(MP_JUCE_HOST_TOOL "" CACHE FILEPATH "Native juceaide executable for cross compilation")
if(MP_JUCE_HOST_TOOL)
  if(NOT JUCE_DIR)
    message(FATAL_ERROR "Cross compilation requires JUCE_DIR")
  endif()
  set(JUCE_MODULES_ONLY ON CACHE BOOL "Use a prebuilt native helper" FORCE)
  add_subdirectory("${JUCE_DIR}" JUCE)
  include("${JUCE_DIR}/extras/Build/CMake/JUCEUtils.cmake")
  add_executable(juce::juceaide IMPORTED GLOBAL)
  set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${MP_JUCE_HOST_TOOL}")
elseif(JUCE_DIR)
  add_subdirectory("${JUCE_DIR}" JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
  FetchContent_MakeAvailable(JUCE)
endif()
FetchContent_Declare(stretch GIT_REPOSITORY https://github.com/Signalsmith-Audio/signalsmith-stretch.git GIT_TAG a670068d9aeb64913331d5cc29337b19a457a7df)
FetchContent_Declare(linear GIT_REPOSITORY https://github.com/Signalsmith-Audio/linear.git GIT_TAG 146b26f91b48835d665fa0498d64d9617818d27a)
FetchContent_MakeAvailable(linear stretch)
add_library(PitchEngine STATIC Source/PitchEngine.cpp)
set_target_properties(PitchEngine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(PitchEngine PUBLIC Source "${JUCE_MODULES_DIR}" PRIVATE ${stretch_SOURCE_DIR} ${linear_SOURCE_DIR}/include)
target_compile_definitions(PitchEngine PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(PitchEngine PRIVATE juce::juce_audio_formats juce::juce_dsp juce::juce_data_structures)
juce_add_plugin(MicrotonalPitchMapPro
  COMPANY_NAME "Microtonal Audio"
  BUNDLE_ID "org.microtonalaudio.pitchmappro"
  PLUGIN_MANUFACTURER_CODE MtAu PLUGIN_CODE MtPm
  PRODUCT_NAME "Microtonal Pitch Map Pro"
  FORMATS VST3 Standalone
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  VST3_CATEGORIES Fx PitchShift
  VST3_AUTO_MANIFEST FALSE)
target_sources(MicrotonalPitchMapPro PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(MicrotonalPitchMapPro PRIVATE PitchEngine juce::juce_audio_utils juce::juce_recommended_config_flags)
target_compile_definitions(MicrotonalPitchMapPro PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
option(MP_BUILD_TESTS "Build DSP and state tests" ON)
if(MP_BUILD_TESTS)
  enable_testing()
  juce_add_console_app(PitchTests PRODUCT_NAME "PitchTests")
  target_sources(PitchTests PRIVATE Tests/PitchTests.cpp)
  target_link_libraries(PitchTests PRIVATE PitchEngine)
  add_test(NAME pitch_engine COMMAND PitchTests)
  juce_add_console_app(PluginTests PRODUCT_NAME "PluginTests")
  target_sources(PluginTests PRIVATE Tests/PluginTests.cpp)
  target_link_libraries(PluginTests PRIVATE MicrotonalPitchMapPro PitchEngine juce::juce_audio_utils)
  target_compile_definitions(PluginTests PRIVATE JUCE_VST3_CAN_REPLACE_VST2=0)
  add_test(NAME plugin_state_and_playback COMMAND PluginTests)
  set_tests_properties(pitch_engine plugin_state_and_playback PROPERTIES TIMEOUT 120)
endif()
