cmake_minimum_required(VERSION 3.24)
project(MicrotonalPitchMapPro VERSION 3.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/bigobj /utf-8)
endif()
include(FetchContent)
set(JUCE_DIR "" CACHE PATH "Existing JUCE 8.0.12 checkout")
set(MP_JUCE_HOST_TOOL "" CACHE FILEPATH "Native juceaide executable for cross compilation")
if(MP_JUCE_HOST_TOOL)
  if(NOT JUCE_DIR)
    message(FATAL_ERROR "Cross compilation requires JUCE_DIR")
  endif()
  set(JUCE_MODULES_ONLY ON CACHE BOOL "Use a prebuilt native helper" FORCE)
  add_subdirectory("${JUCE_DIR}" JUCE)
  include("${JUCE_DIR}/extras/Build/CMake/JUCEUtils.cmake")
  add_executable(juce::juceaide IMPORTED GLOBAL)
  set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${MP_JUCE_HOST_TOOL}")
elseif(JUCE_DIR)
  add_subdirectory("${JUCE_DIR}" JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
  FetchContent_MakeAvailable(JUCE)
endif()
FetchContent_Declare(stretch GIT_REPOSITORY https://github.com/Signalsmith-Audio/signalsmith-stretch.git GIT_TAG a670068d9aeb64913331d5cc29337b19a457a7df)
FetchContent_Declare(linear GIT_REPOSITORY https://github.com/Signalsmith-Audio/linear.git GIT_TAG 146b26f91b48835d665fa0498d64d9617818d27a)
FetchContent_MakeAvailable(linear stretch)
option(MP_ENABLE_ARA "Enable direct host clip access through ARA 2" ON)
set(ARA_SDK_DIR "" CACHE PATH "Existing ARA SDK 2.2.0 checkout")
if(MP_ENABLE_ARA)
  if(NOT ARA_SDK_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ARA_SDK/ARA_API/ARAInterface.h")
    set(ARA_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ARA_SDK")
  endif()
  if(NOT ARA_SDK_DIR)
    FetchContent_Declare(ara_sdk GIT_REPOSITORY https://github.com/Celemony/ARA_SDK.git
      GIT_TAG e723ce48a917b1ce2d7851c78e468f36b51ea151 GIT_SUBMODULES ARA_API ARA_Library SOURCE_SUBDIR no_cmake_build)
    FetchContent_MakeAvailable(ara_sdk)
    set(ARA_SDK_DIR "${ara_sdk_SOURCE_DIR}")
  endif()
  juce_set_ara_sdk_path("${ARA_SDK_DIR}")
endif()
add_library(PitchEngine STATIC Source/PitchEngine.cpp Source/RealtimePitch.cpp Source/Audition.cpp Source/AdaptivePitch.cpp Source/LiveWorkspace.cpp Source/ToneSmoother.cpp Source/FinishFilter.cpp Source/AudioMemory.cpp)
set_target_properties(PitchEngine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(PitchEngine PUBLIC Source "${JUCE_MODULES_DIR}" PRIVATE ${stretch_SOURCE_DIR} ${linear_SOURCE_DIR}/include)
target_compile_definitions(PitchEngine PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(PitchEngine PRIVATE juce::juce_audio_formats juce::juce_dsp juce::juce_data_structures)
juce_add_plugin(MicrotonalPitchMapPro
  COMPANY_NAME "Microtonal Audio"
  BUNDLE_ID "org.microtonalaudio.pitchmappro"
  PLUGIN_MANUFACTURER_CODE MtAu PLUGIN_CODE MtPm
  PRODUCT_NAME "Microtonal Pitch Map Pro"
  FORMATS VST3 Standalone
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  IS_ARA_EFFECT ${MP_ENABLE_ARA}
  ARA_FACTORY_ID "\"org.microtonalaudio.pitchmappro.arafactory\""
  ARA_DOCUMENT_ARCHIVE_ID "\"org.microtonalaudio.pitchmappro.aradoc.v1\""
  VST3_CATEGORIES Fx PitchShift
  VST3_AUTO_MANIFEST FALSE)
target_sources(MicrotonalPitchMapPro PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/ARAIntegration.cpp Source/LiveEditor.cpp)
target_link_libraries(MicrotonalPitchMapPro PRIVATE PitchEngine juce::juce_audio_utils juce::juce_recommended_config_flags)
target_compile_definitions(MicrotonalPitchMapPro PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
option(MP_BUILD_TESTS "Build DSP and state tests" ON)
if(MP_BUILD_TESTS)
  enable_testing()
  juce_add_console_app(PitchTests PRODUCT_NAME "PitchTests")
  target_sources(PitchTests PRIVATE Tests/PitchTests.cpp)
  target_link_libraries(PitchTests PRIVATE PitchEngine)
  add_test(NAME pitch_engine COMMAND PitchTests)
  juce_add_console_app(RegionalTests PRODUCT_NAME "RegionalTests")
  target_sources(RegionalTests PRIVATE Tests/RegionalTests.cpp)
  target_link_libraries(RegionalTests PRIVATE PitchEngine)
  add_test(NAME regional_audio COMMAND RegionalTests)
  set_tests_properties(regional_audio PROPERTIES TIMEOUT 240)
  juce_add_console_app(PluginTests PRODUCT_NAME "PluginTests")
  target_sources(PluginTests PRIVATE Tests/PluginTests.cpp)
  target_link_libraries(PluginTests PRIVATE MicrotonalPitchMapPro PitchEngine)
  target_compile_definitions(PluginTests PRIVATE JUCE_VST3_CAN_REPLACE_VST2=0)
  add_test(NAME plugin_state_and_playback COMMAND PluginTests --audio-only)
  add_test(NAME plugin_editor COMMAND PluginTests)
  juce_add_console_app(WorkflowTests PRODUCT_NAME "WorkflowTests")
  target_sources(WorkflowTests PRIVATE Tests/WorkflowTests.cpp)
  target_link_libraries(WorkflowTests PRIVATE MicrotonalPitchMapPro PitchEngine)
  target_compile_definitions(WorkflowTests PRIVATE JUCE_VST3_CAN_REPLACE_VST2=0)
  add_test(NAME correction_workflow COMMAND WorkflowTests)
  set_tests_properties(correction_workflow PROPERTIES TIMEOUT 120)
  juce_add_console_app(LiveTests PRODUCT_NAME "LiveTests")
  target_sources(LiveTests PRIVATE Tests/LiveTests.cpp)
  target_link_libraries(LiveTests PRIVATE MicrotonalPitchMapPro PitchEngine)
  target_compile_definitions(LiveTests PRIVATE JUCE_VST3_CAN_REPLACE_VST2=0)
  if(MP_ENABLE_ARA)
    target_link_libraries(WorkflowTests PRIVATE juce_ara_sdk)
    target_sources(WorkflowTests PRIVATE "${JUCE_MODULES_DIR}/juce_audio_plugin_client/juce_audio_plugin_client_ARA.cpp")
    target_link_libraries(PluginTests PRIVATE juce_ara_sdk)
    target_link_libraries(LiveTests PRIVATE juce_ara_sdk)
    target_sources(PluginTests PRIVATE "${JUCE_MODULES_DIR}/juce_audio_plugin_client/juce_audio_plugin_client_ARA.cpp")
    target_sources(LiveTests PRIVATE "${JUCE_MODULES_DIR}/juce_audio_plugin_client/juce_audio_plugin_client_ARA.cpp")
    juce_add_console_app(ARATests PRODUCT_NAME "ARATests")
    target_sources(ARATests PRIVATE Tests/ARATests.cpp "${JUCE_MODULES_DIR}/juce_audio_plugin_client/juce_audio_plugin_client_ARA.cpp")
    target_link_libraries(ARATests PRIVATE MicrotonalPitchMapPro PitchEngine juce_ara_sdk)
    target_compile_definitions(ARATests PRIVATE JUCE_VST3_CAN_REPLACE_VST2=0)
    foreach(testTarget PluginTests LiveTests ARATests WorkflowTests)
      target_compile_definitions(${testTarget} PRIVATE JucePlugin_Build_Standalone=1)
    endforeach()
    add_test(NAME ara_clip_transfer COMMAND ARATests)
    set_tests_properties(ara_clip_transfer PROPERTIES TIMEOUT 120)
  endif()
  juce_add_console_app(SmootherTests PRODUCT_NAME "SmootherTests")
  target_sources(SmootherTests PRIVATE Tests/SmootherTests.cpp)
  target_link_libraries(SmootherTests PRIVATE MicrotonalPitchMapPro PitchEngine)
  target_compile_definitions(SmootherTests PRIVATE JUCE_VST3_CAN_REPLACE_VST2=0 JucePlugin_Build_Standalone=1)
  if(MP_ENABLE_ARA)
    target_link_libraries(SmootherTests PRIVATE juce_ara_sdk)
    target_sources(SmootherTests PRIVATE "${JUCE_MODULES_DIR}/juce_audio_plugin_client/juce_audio_plugin_client_ARA.cpp")
  endif()
  add_test(NAME smoother_and_workspace COMMAND SmootherTests)
  set_tests_properties(smoother_and_workspace PROPERTIES TIMEOUT 180)
  add_test(NAME live_audio COMMAND LiveTests)
  set_tests_properties(pitch_engine plugin_state_and_playback plugin_editor live_audio PROPERTIES TIMEOUT 120)
  if(WIN32)
    add_executable(WindowsVst3Audio Tests/WindowsVst3Audio.cpp)
    target_include_directories(WindowsVst3Audio PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
    add_dependencies(WindowsVst3Audio MicrotonalPitchMapPro_VST3)
    add_test(NAME actual_windows_vst3_audio COMMAND WindowsVst3Audio "$<TARGET_FILE:MicrotonalPitchMapPro_VST3>")
    set_tests_properties(actual_windows_vst3_audio PROPERTIES TIMEOUT 120)
  endif()
endif()
